                const canonicalBaseSmell = oldSmell.split('/').slice(-2).join('/');

                CUSTOM_TARGET_SRCS.add(canonicalizeSrc(newSmell));

                const fileKey = oldSmell.split('/').pop();

                    RULES_INDEX.set(fileKey, []);

                RULES_INDEX.get(fileKey).push({
                    oldSmell,
                    phraseLower: phrase.toLowerCase().trim(),
                    newImg: newSmell
                });
            }
        });

        ALL_ORIGINAL_SMELLS = Array.from(originalSmellsSet);
        return rules;
    };

    // произвольную подстроку. Проверяет символы до и после найденного
    // совпадения: если хотя бы один из них — буква, значит это часть более
    // длинного слова, и такое совпадение не засчитывается. Перебираются все
    // вхождения фразы в тексте (через fromIndex), а не только первое —
    // на случай если "неточное" вхождение стоит раньше "точного".
    const hasExactPhrase = (text, phrase) => {
        if (!phrase) return false;
        let fromIndex = 0;
        while (true) {
            const idx = text.indexOf(phrase, fromIndex);
            if (idx === -1) return false;

            const beforeChar = idx > 0 ? text[idx - 1] : '';
            const afterChar = text[idx + phrase.length] || '';

            if (!isWordChar(beforeChar) && !isWordChar(afterChar)) {
                return true;
            }

            fromIndex = idx + 1;
        }
    };

    // normalizedText уже приведён к нижнему регистру один раз в applySmellsToCage.
    // Среди всех подходящих правил выбирается то, у которого фраза длиннее —
    // иначе более общая должность ("матрос") всегда перебивала бы более
    // специфичную ("старший матрос"), если та шла раньше в тексте или просто
    // раньше в списке правил.
    const getMatchingImage = (normalizedText, canonicalBaseSmell) => {
        const fileKey = canonicalBaseSmell.split('/').pop();
        const candidates = RULES_INDEX.get(fileKey);
        if (!candidates) return { newSmell: null, originalBase: null };

        let best = null;
        for (let i = 0; i < candidates.length; i++) {
            const rule = candidates[i];
            if (hasExactPhrase(normalizedText, rule.phraseLower)) {
                if (!best || rule.phraseLower.length > best.phraseLower.length) {
                    best = rule;
                }
            }
        }

        if (best) {
            return { newSmell: best.newImg, originalBase: best.oldSmell };
        }
        return { newSmell: null, originalBase: null };
    };

    const ORIGINAL_SRC_ATTRIBUTE = 'data-original-smell';
    const CURRENT_CAT_ID_ATTRIBUTE = 'data-cat-id';

    const applySmellsToCage = (cage) => {
        const img = cage.querySelector('img[src*="odoroj/"], img[data-original-smell]');
        const catNameElement = cage.querySelector('a');

        if (!img || !catNameElement) return;

        const catUrl = catNameElement.href;
        const idMatch = catUrl.match(/cat(\d+)/);
        const currentCatId = idMatch ? idMatch[1] : null;
        if (!currentCatId) return;

        const textContent = (cage.querySelector('span') || { innerText: '' }).innerText + ' ' + catNameElement.textContent.trim();
        const normalizedText = normalizeText(textContent);

        const currentSrc = img.getAttribute('src');
        const currentCanonical = canonicalizeSrc(currentSrc);
        const cachedCatId = img.getAttribute(CURRENT_CAT_ID_ATTRIBUTE);
        const isCurrentlyCustom = CUSTOM_TARGET_SRCS.has(currentCanonical);

        let originalSrc;

        if (!isCurrentlyCustom) {
            // Сейчас в DOM реально ванильная (не наша) картинка — это самый надёжный
            // момент, чтобы зафиксировать её как оригинал. Перезаписываем сохранённое
            // значение каждый раз, когда видим такую ситуацию, а не только один раз:
            // так при смене локации/кота в этой же клетке новый настоящий запах
            // подхватится сразу, а не будет проигнорирован из-за старого кэша.
            originalSrc = currentSrc;
            img.setAttribute(ORIGINAL_SRC_ATTRIBUTE, originalSrc);
            img.setAttribute(CURRENT_CAT_ID_ATTRIBUTE, currentCatId);
        } else if (cachedCatId === currentCatId && img.getAttribute(ORIGINAL_SRC_ATTRIBUTE)) {
            // Сейчас показана наша же подмена, и кот в клетке тот же, для которого
            // мы её и делали, — доверяем последнему зафиксированному оригиналу.
            originalSrc = img.getAttribute(ORIGINAL_SRC_ATTRIBUTE);
        } else {
            // Показана какая-то подмена, но либо кот сменился, либо надёжного
            // оригинала для него ещё нет (например DOM-нода переиспользована Vue
            // под другого кота, и framework пока не успел сбросить src на
            // ванильный). Без похода на сервер точно оригинал не узнать — берём
            // текущую картинку как временный ориентир; как только сайт покажет
            // настоящий ванильный вид, это самоисправится веткой выше.
            originalSrc = currentSrc;
            img.setAttribute(ORIGINAL_SRC_ATTRIBUTE, originalSrc);
            img.setAttribute(CURRENT_CAT_ID_ATTRIBUTE, currentCatId);
        }

        const originalSrcPart = canonicalizeSrc(originalSrc);
        const matchingResult = getMatchingImage(normalizedText, originalSrcPart);
        const targetSrc = matchingResult.newSmell || originalSrc;

        if (canonicalizeSrc(targetSrc) !== currentCanonical) {
            img.src = targetSrc;
        }
    };

    const initSmellObservers = () => {
        loadData();

        const mapContainer = document.querySelector('#ist, #cages_div');
        if (!mapContainer) {
             return;
        }

        // Вместо отдельного MutationObserver на каждую клетку (может быть сотни на
        // карте) используем один общий наблюдатель на весь контейнер. Изменения
        // собираются в очередь и обрабатываются одним проходом за кадр анимации,
        // чтобы не пересчитывать всё при каждом мелком чихе DOM.
        const pendingCages = new Set();
        let flushScheduled = false;

        const flushPending = () => {
            flushScheduled = false;
            pendingCages.forEach(cage => {
                if (cage.isConnected) {
                    applySmellsToCage(cage);
                }
            });
            pendingCages.clear();
        };

        const scheduleFlush = () => {
            if (flushScheduled) return;
            flushScheduled = true;
            requestAnimationFrame(flushPending);
        };

        // Первичное применение ко всем уже существующим клеткам.
        document.querySelectorAll('.cage').forEach(cage => applySmellsToCage(cage));

        const mapChangeObserver = new MutationObserver((mutationsList) => {
            for (const mutation of mutationsList) {
                if (mutation.type === 'childList') {
                    mutation.addedNodes.forEach(node => {
                        if (node.nodeType !== 1) return;
                        if (node.classList && node.classList.contains('cage')) {
                            pendingCages.add(node);
                        } else if (node.querySelectorAll) {
                            node.querySelectorAll('.cage').forEach(cage => pendingCages.add(cage));
                        }
                    });
                }

                const target = mutation.target;
                const cage = target && target.nodeType === 1
                    ? target.closest('.cage')
                    : (target && target.parentElement ? target.parentElement.closest('.cage') : null);

                if (cage) {
                    if (mutation.type === 'childList' || mutation.type === 'characterData') {
                        pendingCages.add(cage);
                    } else if (mutation.type === 'attributes' && mutation.attributeName === 'href') {
                        pendingCages.add(cage);
                    }
                }
            }

            if (pendingCages.size > 0) {
                scheduleFlush();
            }
        });

        mapChangeObserver.observe(mapContainer, {
            childList: true,
            subtree: true,
            characterData: true,
            attributes: true,
            attributeFilter: ['href']
        });

        // Подстраховка на случай, если MutationObserver пропустил смену кота в клетке
        // (Vue может патчить/переиспользовать DOM-узлы так, что наш фильтр мутаций не
        // срабатывает). Перепроверяем запах в момент, когда курсор входит в клетку —
        // то есть ровно тогда, когда должен открыться .cat_tooltip. Картинка меняется
        // мгновенно, так что это незаметно.
        //
        // Один делегированный слушатель на весь контейнер (не по одному на клетку),
        // плюс дедуп по lastHoveredCage — applySmellsToCage() вызывается только при
        // переходе на новую клетку, а не на каждое микро-движение мыши внутри одной
        // и той же клетки (где события mouseover иначе сыпались бы десятками, так как
        // внутри .cat_tooltip много вложенных элементов).
        let lastHoveredCage = null;

        const recheckOnEnter = (target) => {
            const cage = target.closest ? target.closest('.cage') : null;
            if (cage && cage !== lastHoveredCage) {
                lastHoveredCage = cage;
                applySmellsToCage(cage);
            }
        };

        mapContainer.addEventListener('mouseover', (e) => {
            recheckOnEnter(e.target);
        }, { passive: true });

        mapContainer.addEventListener('mouseout', (e) => {
            if (lastHoveredCage && !lastHoveredCage.contains(e.relatedTarget)) {
                lastHoveredCage = null;
            }
        }, { passive: true });

        // На тач-устройствах тултип обычно открывается тапом, а не наведением —
        // та же перепроверка на touchstart (дедуп тут не критичен, событие и так
        // не «дребезжит» как mouseover).
        mapContainer.addEventListener('touchstart', (e) => {
            recheckOnEnter(e.target);
        }, { passive: true });

        window.smellObservers = [mapChangeObserver];
    };

                    sectionRules.appendChild(item);
                });

                details.appendChild(sectionRules);
                ruleList.appendChild(details);
            });
        };
        const collectData = () => {
            const data = [];
            panel.querySelectorAll('.rule-item').forEach(item => {
                const oldSmell = item.querySelector('.old-smell').value.trim();
                const phrase = item.querySelector('.phrase').value.trim();
                const image = item.querySelector('.image').value.trim();
                if (oldSmell && phrase && image) data.push([oldSmell, phrase, image]);
            });
            return data;
        };

        // Какие секции сейчас раскрыты, чтобы не сворачивать их пользователю
        // при перерисовке списка (например после удаления строки).
        const getOpenSectionNames = () => {
            const open = new Set();
            ruleList.querySelectorAll('.rule-section[open]').forEach(details => {
                open.add(details.dataset.sectionName);
            });
            return open;
        };

        // Ненавязчивая подсказка вместо alert/confirm: показывается на пару секунд
        // в правом верхнем углу панели и сама плавно скрывается.
        const toast = panel.querySelector('#sp-toast');
        const toastText = toast.querySelector('.sp-toast-text');
        let toastHideTimer = null;
        const showToast = (message) => {
            toastText.textContent = message;
            clearTimeout(toastHideTimer);
            toast.classList.add('show');
            toastHideTimer = setTimeout(() => {
                toast.classList.remove('show');
            }, 2200);
        };

        addBtn.onclick = () => {
            const newData = collectData();
            newData.push(["", "", ""]);
            const openSections = getOpenSectionNames();
            openSections.add('Другое');
            renderRules(newData, openSections);
        };
        ruleList.onclick = (e) => {
            if (e.target.closest('.row-remove')) {
                e.target.closest('.rule-item').remove();
                // Перерисовываем весь список: так счётчик строк в заголовке секции
                // обновляется сразу, а опустевшая секция пропадает, а не остаётся
                // висеть пустой оболочкой.
                renderRules(collectData(), getOpenSectionNames());
            }
        };
        saveBtn.onclick = () => {
            const dataToSave = collectData();
            saveData(dataToSave);
            showToast('Сохранено');
        };
        resetBtn.onclick = () => {
            if (confirm("Сбросить запахи на дефолтные?")) {
                resetData();
                renderRules(DEFAULT_RULES);
                showToast('Сброшено на дефолтные');
            }
        };
        deleteAllBtn.onclick = () => {
            if (!confirm("Удалить все запахи без возможности отмены (кроме повторного добавления вручную)?")) return;
            renderRules([]);
            gmSetValueSync(STORAGE_KEY, "[]");
            showToast('Все запахи удалены');
        };

        const updateBtn = panel.querySelector('#check-update-btn');
        const updateLabel = updateBtn.querySelector('.sp-update-label');
        const currentVersion = (typeof GM_info === 'object' && GM_info.script && GM_info.script.version)
            ? GM_info.script.version
            : null;

        updateBtn.onclick = () => {
            if (updateBtn.disabled) return;
            updateBtn.disabled = true;
            updateLabel.textContent = 'Проверяем...';

            fetchRemoteVersion()
                .then((remoteVersion) => {
                    if (!currentVersion) {
                        showToast(`Актуальная версия: ${remoteVersion}`);
                        return;
                    }
                    if (isVersionNewer(remoteVersion, currentVersion)) {
                        showToast(`Найдено обновление v${remoteVersion}, открываю установку...`);
                        setTimeout(() => openUrlInNewTab(DOWNLOAD_URL), 600);
                    } else {
                        showToast('У вас последняя версия');
                    }
                })
                .catch((err) => {
                    console.error('Проверка обновлений не удалась.', err);
                    showToast('Не удалось проверить обновление');
                })
                .finally(() => {
                    updateBtn.disabled = false;
                    updateLabel.textContent = 'Проверить обновление';
                });
        };

        renderRules(currentData);
    };

    const waitForElement = (selector) => new Promise(resolve => {
        const element = document.querySelector(selector);
        if (element) return resolve(element);
        const observer = new MutationObserver((_, obs) => {
            const el = document.querySelector(selector);
            if (el) {
                obs.disconnect();
                resolve(el);
            }
        });
        observer.observe(document.body, { childList: true, subtree: true });
    });

    if (window.location.pathname.endsWith('/settings')) {
        waitForElement('#site_table').then(createSettingsInterface);
    } else if (document.querySelector('#main_table')) {
        waitForElement('#ist, #cages_div').then(initSmellObservers);
    }
})();
