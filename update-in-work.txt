// ==UserScript==
// @name         CW smell (DEBUG PANEL)
// @namespace    http://tampermonkey.net/
// @version      1.0.17-debug1
// @description  Временная отладочная сборка с видимой панелью логов для диагностики на мобильных браузерах.
// @author       achterstem
// @match        http*://*.catwar.net/*
// @match        http*://*.catwar.su/*
// @icon         https://www.google.com/s2/favicons?sz=64&domain=catwar.su
// @license      MIT
// @grant        GM_setValue
// @grant        GM_getValue
// @grant        GM_deleteValue
// @grant        GM_listValues
// @run-at       document-idle
// ==/UserScript==

(function() {
    'use strict';

    // ==================== ОТЛАДОЧНАЯ ПАНЕЛЬ ====================
    // Показывает логи прямо на странице, потому что на телефоне обычно нет
    // доступа к консоли разработчика. Не влияет на основную логику ниже.
    const LOG_LINES = [];
    const MAX_LOG_LINES = 300;
    let logPanelEl = null;
    let logTextEl = null;

    const renderLog = () => {
        if (!logTextEl) return;
        logTextEl.textContent = LOG_LINES.join('\n');
        logTextEl.scrollTop = logTextEl.scrollHeight;
    };

    const dlog = (msg) => {
        const line = `[${new Date().toISOString().substr(11, 8)}] ${msg}`;
        LOG_LINES.push(line);
        if (LOG_LINES.length > MAX_LOG_LINES) LOG_LINES.shift();
        renderLog();
        try { console.log('[CW-smell-debug]', msg); } catch (e) {}
    };

    const createDebugPanel = () => {
        const style = document.createElement('style');
        style.innerHTML = `
            #cwsmell-debug-panel {
                position: fixed; left: 4px; right: 4px; bottom: 4px; z-index: 2147483647;
                max-height: 40vh; background: rgba(10,10,10,0.92); color: #7CFC00;
                font-family: monospace; font-size: 10px; line-height: 1.35;
                border: 1px solid #444; border-radius: 6px; display: flex; flex-direction: column;
            }
            #cwsmell-debug-panel .cwsd-header {
                display: flex; gap: 6px; padding: 4px 6px; background: #1a1a1a; border-bottom: 1px solid #444;
                align-items: center; flex-wrap: wrap;
            }
            #cwsmell-debug-panel .cwsd-header span { color: #ccc; font-size: 10px; margin-right: auto; }
            #cwsmell-debug-panel button {
                font-size: 10px; padding: 3px 8px; border-radius: 4px; border: 1px solid #666;
                background: #2a2a2a; color: #eee;
            }
            #cwsmell-debug-panel .cwsd-body {
                overflow-y: auto; padding: 4px 6px; white-space: pre-wrap; word-break: break-all;
            }
            #cwsmell-debug-panel.collapsed .cwsd-body { display: none; }
        `;
        document.head.appendChild(style);

        const panel = document.createElement('div');
        panel.id = 'cwsmell-debug-panel';
        panel.innerHTML = `
            <div class="cwsd-header">
                <span>CW-smell DEBUG</span>
                <button type="button" id="cwsd-copy">Копировать</button>
                <button type="button" id="cwsd-clear">Очистить</button>
                <button type="button" id="cwsd-toggle">Свернуть</button>
            </div>
            <div class="cwsd-body"></div>
        `;
        document.documentElement.appendChild(panel);

        logPanelEl = panel;
        logTextEl = panel.querySelector('.cwsd-body');

        panel.querySelector('#cwsd-clear').onclick = () => {
            LOG_LINES.length = 0;
            renderLog();
        };
        panel.querySelector('#cwsd-toggle').onclick = () => {
            panel.classList.toggle('collapsed');
        };
        panel.querySelector('#cwsd-copy').onclick = () => {
            const text = LOG_LINES.join('\n');
            const finish = (ok) => {
                const btn = panel.querySelector('#cwsd-copy');
                const old = btn.textContent;
                btn.textContent = ok ? 'Скопировано!' : 'Не вышло, выделите вручную';
                setTimeout(() => { btn.textContent = old; }, 1500);
            };
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).then(() => finish(true)).catch(() => finish(false));
            } else {
                try {
                    const ta = document.createElement('textarea');
                    ta.value = text;
                    ta.style.position = 'fixed';
                    ta.style.opacity = '0';
                    document.body.appendChild(ta);
                    ta.focus();
                    ta.select();
                    document.execCommand('copy');
                    document.body.removeChild(ta);
                    finish(true);
                } catch (e) {
                    finish(false);
                }
            }
        };

        renderLog();
    };

    // Ловим ошибки, которые случаются вне наших try/catch (например если
    // что-то на странице сломается из-за нашего кода) — иначе мы их не увидим.
    window.addEventListener('error', (e) => {
        dlog(`GLOBAL ERROR: ${e.message} @ ${e.filename}:${e.lineno}`);
    });

    // ==================== ОСНОВНАЯ ЛОГИКА (как в рабочей версии) ====================

    const STORAGE_KEY = 'CUSTOM_SMELLS_DATA';

    const gmGetValueSync = (key, defaultValue) => {
        if (typeof GM_getValue === 'function') {
            try { return GM_getValue(key, defaultValue); } catch (e) { dlog('gmGetValue error: ' + e.message); }
        }
        return defaultValue;
    };

    const DEFAULT_CATEGORIES = [
        {
            name: 'ЗАСТЫВШАЯ ЭПОХА',
            rules: [
                ['232', 'Провидец Времени', '432'],
                ['232', 'Наследник', '432'],
                ['232', 'Верховодец', '432'],
                ['232', 'Старший Отец', '432'],
            ]
        },
        {
            name: 'ОЗХ',
            rules: [
                ['150', 'Вождь', '452'],
                ['150', 'Малютка', '452'],
            ]
        },
    ];
    // ВНИМАНИЕ: в этой отладочной сборке DEFAULT_CATEGORIES сокращён до нескольких
    // тестовых правил специально, чтобы упростить чтение логов. Полный список
    // правил из рабочей версии сюда не переносился — эта сборка только для
    // диагностики механики (находит ли скрипт клетки, ловит ли события, меняется
    // ли src), а не для реального использования на карте.

    const DEFAULT_RULES = DEFAULT_CATEGORIES.reduce((acc, cat) => acc.concat(cat.rules), []);

    const normalizeSmellValue = (value) => {
        if (!value) return value;
        const trimmed = value.trim();
        if (/^\d+$/.test(trimmed)) return `odoroj/${trimmed}.png`;
        return trimmed;
    };

    const shortenSmellValue = (value) => {
        if (!value) return value;
        const trimmed = value.trim();
        const match = /^odoroj\/(\d+)\.png$/.exec(trimmed);
        return match ? match[1] : trimmed;
    };

    const canonicalizeSrc = (src) => {
        if (!src) return src;
        return src.split('/').slice(-2).join('/');
    };

    let RULES_INDEX = new Map();
    let CUSTOM_TARGET_SRCS = new Set();

    const loadData = () => {
        let storedData = gmGetValueSync(STORAGE_KEY, null);
        let rules;
        try {
            rules = storedData ? JSON.parse(storedData) : DEFAULT_RULES;
        } catch (e) {
            dlog('loadData parse error: ' + e.message);
            rules = DEFAULT_RULES;
        }
        rules = rules.map(([o, p, n]) => [shortenSmellValue(o), p, shortenSmellValue(n)]);

        RULES_INDEX.clear();
        CUSTOM_TARGET_SRCS.clear();

        rules.forEach(([oldSmellRaw, phrase, newSmellRaw]) => {
            if (oldSmellRaw && newSmellRaw) {
                const oldSmell = normalizeSmellValue(oldSmellRaw);
                const newSmell = normalizeSmellValue(newSmellRaw);
                CUSTOM_TARGET_SRCS.add(canonicalizeSrc(newSmell));
                const fileKey = oldSmell.split('/').pop();
                if (!RULES_INDEX.has(fileKey)) RULES_INDEX.set(fileKey, []);
                RULES_INDEX.get(fileKey).push({ oldSmell, phraseLower: phrase.toLowerCase().trim(), newImg: newSmell });
            }
        });

        dlog(`loadData: правил=${rules.length}, групп RULES_INDEX=${RULES_INDEX.size}`);
        return rules;
    };

    const normalizeText = (text) => text.toLowerCase().replace(/\s+/g, ' ').trim();
    const isWordChar = (ch) => !!ch && /[a-zа-яёA-ZА-ЯЁ]/.test(ch);

    const hasExactPhrase = (text, phrase) => {
        if (!phrase) return false;
        let fromIndex = 0;
        while (true) {
            const idx = text.indexOf(phrase, fromIndex);
            if (idx === -1) return false;
            const beforeChar = idx > 0 ? text[idx - 1] : '';
            const afterChar = text[idx + phrase.length] || '';
            if (!isWordChar(beforeChar) && !isWordChar(afterChar)) return true;
            fromIndex = idx + 1;
        }
    };

    const getMatchingImage = (normalizedText, canonicalBaseSmell) => {
        const fileKey = canonicalBaseSmell.split('/').pop();
        const candidates = RULES_INDEX.get(fileKey);
        if (!candidates) return { newSmell: null };
        let best = null;
        for (const rule of candidates) {
            if (hasExactPhrase(normalizedText, rule.phraseLower)) {
                if (!best || rule.phraseLower.length > best.phraseLower.length) best = rule;
            }
        }
        return best ? { newSmell: best.newImg, matchedPhrase: best.phraseLower } : { newSmell: null };
    };

    const ORIGINAL_SRC_ATTRIBUTE = 'data-original-smell';
    const CURRENT_CAT_ID_ATTRIBUTE = 'data-cat-id';
    const seenImgElements = new WeakSet();

    const applySmellsToCage = (cage, source) => {
        try {
            const img = cage.querySelector('img[src*="odoroj/"], img[data-original-smell]');
            const catNameElement = cage.querySelector('a');

            if (!img) { dlog(`[${source}] клетка без подходящего <img> (селектор не нашёл ничего)`); return; }
            if (!catNameElement) { dlog(`[${source}] клетка без <a> (нет имени/ссылки на кота)`); return; }

            const catUrl = catNameElement.href;
            const idMatch = catUrl.match(/cat(\d+)/);
            const currentCatId = idMatch ? idMatch[1] : null;
            if (!currentCatId) { dlog(`[${source}] не нашли catId в href="${catUrl}"`); return; }

            if (!seenImgElements.has(img)) {
                seenImgElements.add(img);
                dlog(`[${source}] новая карточка: catId=${currentCatId}, img.src="${img.getAttribute('src')}"`);
            }

            const textContent = (cage.querySelector('span') || { innerText: '' }).innerText + ' ' + catNameElement.textContent.trim();
            const normalizedText = normalizeText(textContent);

            const currentSrc = img.getAttribute('src');
            const currentCanonical = canonicalizeSrc(currentSrc);
            const cachedCatId = img.getAttribute(CURRENT_CAT_ID_ATTRIBUTE);
            const isCurrentlyCustom = CUSTOM_TARGET_SRCS.has(currentCanonical);

            let originalSrc;
            let branch;

            if (!isCurrentlyCustom) {
                originalSrc = currentSrc;
                img.setAttribute(ORIGINAL_SRC_ATTRIBUTE, originalSrc);
                img.setAttribute(CURRENT_CAT_ID_ATTRIBUTE, currentCatId);
                branch = 'ванильный, зафиксировали';
            } else if (cachedCatId === currentCatId && img.getAttribute(ORIGINAL_SRC_ATTRIBUTE)) {
                originalSrc = img.getAttribute(ORIGINAL_SRC_ATTRIBUTE);
                branch = 'наша подмена, тот же кот, берём кэш';
            } else {
                originalSrc = currentSrc;
                img.setAttribute(ORIGINAL_SRC_ATTRIBUTE, originalSrc);
                img.setAttribute(CURRENT_CAT_ID_ATTRIBUTE, currentCatId);
                branch = 'подмена + кот сменился, временный ориентир';
            }

            const originalSrcPart = canonicalizeSrc(originalSrc);
            const matchingResult = getMatchingImage(normalizedText, originalSrcPart);
            const targetSrc = matchingResult.newSmell || originalSrc;

            const willChange = canonicalizeSrc(targetSrc) !== currentCanonical;

            dlog(`[${source}] catId=${currentCatId} текст="${normalizedText}" ветка=(${branch}) ориг=${originalSrcPart} совпадение=${matchingResult.matchedPhrase || 'НЕТ'} target=${canonicalizeSrc(targetSrc)} меняем=${willChange}`);

            if (willChange) {
                img.src = targetSrc;
                dlog(`[${source}] >>> img.src изменён на ${targetSrc}`);
            }
        } catch (e) {
            dlog(`[${source}] ИСКЛЮЧЕНИЕ в applySmellsToCage: ${e.message}`);
        }
    };

    const initSmellObservers = () => {
        loadData();

        const mapContainer = document.querySelector('#ist, #cages_div');
        if (!mapContainer) {
            dlog('initSmellObservers: контейнер #ist/#cages_div НЕ найден');
            return;
        }
        dlog(`initSmellObservers: контейнер найден, id="${mapContainer.id}"`);

        const allCages = document.querySelectorAll('.cage');
        dlog(`initSmellObservers: найдено .cage элементов = ${allCages.length}`);
        allCages.forEach(cage => applySmellsToCage(cage, 'init'));

        const pendingCages = new Set();
        let flushScheduled = false;

        const flushPending = () => {
            flushScheduled = false;
            const n = pendingCages.size;
            pendingCages.forEach(cage => {
                if (cage.isConnected) applySmellsToCage(cage, 'mutation');
            });
            pendingCages.clear();
            if (n > 0) dlog(`flushPending: обработано клеток = ${n}`);
        };

        const scheduleFlush = () => {
            if (flushScheduled) return;
            flushScheduled = true;
            requestAnimationFrame(flushPending);
        };

        const mapChangeObserver = new MutationObserver((mutationsList) => {
            for (const mutation of mutationsList) {
                if (mutation.type === 'childList') {
                    mutation.addedNodes.forEach(node => {
                        if (node.nodeType !== 1) return;
                        if (node.classList && node.classList.contains('cage')) {
                            pendingCages.add(node);
                        } else if (node.querySelectorAll) {
                            node.querySelectorAll('.cage').forEach(cage => pendingCages.add(cage));
                        }
                    });
                }
                const target = mutation.target;
                const cage = target && target.nodeType === 1
                    ? target.closest('.cage')
                    : (target && target.parentElement ? target.parentElement.closest('.cage') : null);
                if (cage) {
                    if (mutation.type === 'childList' || mutation.type === 'characterData') {
                        pendingCages.add(cage);
                    } else if (mutation.type === 'attributes' && mutation.attributeName === 'href') {
                        pendingCages.add(cage);
                    }
                }
            }
            if (pendingCages.size > 0) scheduleFlush();
        });

        mapChangeObserver.observe(mapContainer, {
            childList: true, subtree: true, characterData: true,
            attributes: true, attributeFilter: ['href']
        });
        dlog('MutationObserver подключён к контейнеру');

        let lastHoveredCage = null;
        const recheckOnEnter = (target, source) => {
            const cage = target.closest ? target.closest('.cage') : null;
            if (!cage) {
                dlog(`${source}: ближайшего .cage не найдено для target=${target.tagName}.${target.className}`);
                return;
            }
            if (cage !== lastHoveredCage) {
                lastHoveredCage = cage;
                applySmellsToCage(cage, source);
            }
        };

        mapContainer.addEventListener('mouseover', (e) => recheckOnEnter(e.target, 'mouseover'), { passive: true });
        mapContainer.addEventListener('mouseout', (e) => {
            if (lastHoveredCage && !lastHoveredCage.contains(e.relatedTarget)) lastHoveredCage = null;
        }, { passive: true });
        mapContainer.addEventListener('touchstart', (e) => recheckOnEnter(e.target, 'touchstart'), { passive: true });

        dlog('Слушатели mouseover/mouseout/touchstart подключены');
    };

    const waitForElement = (selector) => new Promise(resolve => {
        const element = document.querySelector(selector);
        if (element) return resolve(element);
        const observer = new MutationObserver((_, obs) => {
            const el = document.querySelector(selector);
            if (el) { obs.disconnect(); resolve(el); }
        });
        observer.observe(document.body, { childList: true, subtree: true });
    });

    createDebugPanel();
    dlog(`Скрипт загружен. path=${window.location.pathname}, UA=${navigator.userAgent}`);

    if (window.location.pathname.endsWith('/settings')) {
        dlog('Страница /settings — отладочная сборка не рендерит панель настроек, только диагностику карты.');
    } else if (document.querySelector('#main_table')) {
        dlog('#main_table найден, ждём #ist/#cages_div...');
        waitForElement('#ist, #cages_div').then((el) => {
            dlog(`#ist/#cages_div появился: id="${el.id}"`);
            initSmellObservers();
        });
    } else {
        dlog('#main_table НЕ найден на этой странице — initSmellObservers не запускается.');
    }
})();
